    document.addEventListener("DOMContentLoaded", function() {
        
        // --- 1. TEXT ANIMATION ---
        gsap.to(".j-hero-title", { opacity: 1, y: 0, duration: 1, ease: "power3.out", delay: 0.5 });
        gsap.to(".j-hero-subtitle", { opacity: 1, y: 0, duration: 1, ease: "power3.out", delay: 0.8 });
        gsap.to(".j-hero-btn", { opacity: 1, y: 0, duration: 1, ease: "power3.out", delay: 1.1 });

        // --- 2. BACKGROUND SLIDER WITH KEN BURNS ---
        const slides = gsap.utils.toArray(".j-slide"); 
        let currentSlide = slides[0]; 

        gsap.set(slides[0], { autoAlpha: 1 }); 
        gsap.set(slides.slice(1), { autoAlpha: 0 });

        function rotateSlides() {
            let nextSlide = slides[(slides.indexOf(currentSlide) + 1) % slides.length];
            
            let currentImage = currentSlide.querySelector('.j-image-bg');
            let nextImage = nextSlide.querySelector('.j-image-bg');

            let tl = gsap.timeline({
                onComplete: () => {
                    currentSlide = nextSlide;
                    gsap.delayedCall(4, rotateSlides); 
                }
            });

            if (currentImage) { gsap.set(currentImage, { scale: 1 }); }

            tl.to(currentSlide, { autoAlpha: 0, duration: 2, ease: "power2.inOut" })
              .to(nextSlide, { autoAlpha: 1, duration: 2, ease: "power2.inOut" }, "<");

            if (nextImage) {
                gsap.set(nextImage, { scale: 1 });
                tl.to(nextImage, { scale: 1.15, duration: 7, ease: "none" }, "<"); 
            }
        }

        gsap.delayedCall(4, rotateSlides);
    });
</script>

<script>
    document.addEventListener("DOMContentLoaded", function() {
        const slider = document.getElementById("jScrollContainer");
        
        // Configuration
        // 0.05 is "heavy/slow", 0.1 is "snappy/fast"
        const ease = 0.06; 
        
        let currentScroll = 0;
        let targetScroll = 0;
        let isDragging = false;
        let startX;
        let startScrollLeft;
        let maxScroll = 0;

        // Initialize
        function init() {
            maxScroll = slider.scrollWidth - slider.clientWidth;
            targetScroll = slider.scrollLeft;
            currentScroll = slider.scrollLeft;
            animate();
        }

        // LERP Animation Loop (The "Lenis" Magic)
        function animate() {
            // Formula: current = current + (target - current) * ease
            // This creates the smooth "catch up" effect
            currentScroll += (targetScroll - currentScroll) * ease;
            
            // Apply the scroll
            slider.scrollLeft = currentScroll;
            
            // Recalculate maxScroll in case window resized
            maxScroll = slider.scrollWidth - slider.clientWidth;

            // Keep loop running
            requestAnimationFrame(animate);
        }

        // 1. Mouse Wheel Logic
        slider.addEventListener("wheel", (e) => {
            // Prevent vertical page scroll ONLY if we have room to scroll horizontally
            if (
                (e.deltaY > 0 && targetScroll < maxScroll - 1) || 
                (e.deltaY < 0 && targetScroll > 1)
            ) {
                e.preventDefault();
            }

            // Add momentum to target
            targetScroll += e.deltaY * 1.5; // Multiplier for speed
            
            // Clamp target within bounds
            targetScroll = Math.max(0, Math.min(targetScroll, maxScroll));
        }, { passive: false });


        // 2. Touch/Drag Logic (Hybrid)
        // This ensures it works great on phones or if you click+drag with mouse
        slider.addEventListener('mousedown', (e) => {
            isDragging = true;
            slider.classList.add('is-dragging');
            startX = e.pageX - slider.offsetLeft;
            startScrollLeft = targetScroll;
        });

        slider.addEventListener('mouseleave', () => {
            isDragging = false;
            slider.classList.remove('is-dragging');
        });

        slider.addEventListener('mouseup', () => {
            isDragging = false;
            slider.classList.remove('is-dragging');
        });

        slider.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            e.preventDefault();
            const x = e.pageX - slider.offsetLeft;
            const walk = (x - startX) * 2; // Drag speed
            targetScroll = startScrollLeft - walk;
            targetScroll = Math.max(0, Math.min(targetScroll, maxScroll));
        });

        // Initialize on load

        });

        // 4. TEXT REVEAL (FIXED)
        // Set 'start' to 'top 90%' so it triggers earlier
        // Added 'clearProps' to ensure text is visible even if animation glitches
        gsap.from([".j-app-headline", ".j-app-desc", ".j-app-buttons"], {
            y: 40, 
            opacity: 0, 
            duration: 1, 
            stagger: 0.15, 
            ease: "power3.out",
            scrollTrigger: { 
                trigger: "#jaguar-section-6", 
                start: "top 90%", // Trigger almost immediately when section appears
            },
            onComplete: () => {
                gsap.set([".j-app-headline", ".j-app-desc", ".j-app-buttons"], { clearProps: "opacity" });
            }
        });

        // 5. PHONE ENTRANCE
        gsap.from(".j-phone-wrapper", {
            y: 100, opacity: 0, duration: 1.4, ease: "power3.out",
            scrollTrigger: { trigger: "#jaguar-section-6", start: "top 85%" }
        });

        // 6. FLOAT
        gsap.to(".j-phone-tilt-layer", {
            y: -25, duration: 4, repeat: -1, yoyo: true, ease: "sine.inOut"
        });

        // 7. MOUSE TILT
        const section = document.querySelector("#jaguar-section-6");
        const tiltLayer = document.querySelector(".j-phone-tilt-layer");
        const glow = document.querySelector(".j-app-glow");

        section.addEventListener("mousemove", (e) => {
            if (ScrollTrigger.isScrolling()) return;
            const { width, height, left, top } = section.getBoundingClientRect();
            const xPos = (e.clientX - left - width / 2) / (width / 2);
            const yPos = (e.clientY - top - height / 2) / (height / 2);

            gsap.to(tiltLayer, { rotationY: xPos * 15, rotationX: -yPos * 15, duration: 0.5, ease: "power2.out" });
            gsap.to(glow, { x: -xPos * 60, y: -yPos * 60, duration: 1, ease: "power2.out" });
        });

        section.addEventListener("mouseleave", () => {
            gsap.to(tiltLayer, { rotationY: 0, rotationX: 0, duration: 1, ease: "power2.out" });
            gsap.to(glow, { x: 0, y: 0, duration: 1, ease: "power2.out" });
        });
    });
</script>
