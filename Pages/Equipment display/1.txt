    <script>
        // Select all hotspots and cards
        const hotspots = document.querySelectorAll('.hotspot');
        const cards = document.querySelectorAll('.spec-card');

        // Function to close all cards (reset state)
        function closeAllCards() {
            cards.forEach(card => card.classList.remove('active'));
            hotspots.forEach(spot => spot.classList.remove('active'));
        }

        // Function to position the card relative to the hotspot
        function positionCard(hotspot, card) {
            // Get coordinates
            const spotRect = hotspot.getBoundingClientRect();
            const wrapperRect = hotspot.parentElement.getBoundingClientRect();
            
            // Calculate top/left relative to the wrapper
            let top = spotRect.top - wrapperRect.top;
            let left = spotRect.left - wrapperRect.left;
            
            // Card dimensions (based on CSS)
            const cardWidth = 240; 
            
            // Logic: Place to the right, unless it goes off screen, then flip left.
            
            // Check remaining space on the right
            const remainingSpaceRight = wrapperRect.width - left;

            if (remainingSpaceRight < cardWidth + 40) {
                // Not enough space on right, flip to left
                card.style.left = (left - cardWidth - 10) + 'px';
                card.style.transformOrigin = "right center";
            } else {
                // Default: Place to right
                card.style.left = (left + 40) + 'px'; 
                card.style.transformOrigin = "left center";
            }
            
            // Vertical alignment: center slightly up from the dot
            card.style.top = (top - 20) + 'px';
        }

        hotspots.forEach(hotspot => {
            // Click/Tap Event
            hotspot.addEventListener('click', (e) => {
                e.stopPropagation(); // Prevent closing immediately by bubbling to wrapper
                
                const targetId = hotspot.getAttribute('data-target');
                const targetCard = document.getElementById(targetId);
                
                // If this card is already open, close it (toggle behavior)
                if (targetCard.classList.contains('active')) {
                    closeAllCards();
                } else {
                    // Close others first
                    closeAllCards();
                    
                    // Position and Open new one
                    positionCard(hotspot, targetCard);
                    
                    // Small timeout to allow position to set before animating opacity
                    requestAnimationFrame(() => {
                        targetCard.classList.add('active');
                        hotspot.classList.add('active');
                    });
                }
            });
        });

        // Close tooltip if user clicks anywhere else on the wrapper/image
        document.querySelector('.blueprint-wrapper').addEventListener('click', () => {
            closeAllCards();
        });
    </script>
